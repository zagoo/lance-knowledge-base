   52 +/// Bytes hashbrown allocates for the `HashMap<u64, Option<u64>>` that
   53 +/// `HashMap::with_capacity(entries)` reserves.
   54 +///
   55 +/// hashbrown's `capacity_to_buckets` rounds the capacity up to
   56 +/// `next_power_of_two(ceil(entries * 8 / 7))` buckets (4 below four entries, 8
   57 +/// below eight) and lays the table out as one slot of
   58 +/// `size_of::<(u64, Option<u64>)>()` (24) bytes plus one control byte per
   59 +/// bucket, followed by one group width (16) of trailing control bytes. The
   60 +/// rounding is what a flat per-entry charge misses: 7,350,000 entries adjust
   61 +/// to 8,400,000 and round to 16,777,216 buckets, 419,430,416 bytes, where 32
   62 +/// bytes per entry would have charged 235,200,000.
   63 +fn hash_map_bytes(entries: u64) -> u64 {
   64 +    const SLOT_BYTES: u64 = std::mem::size_of::<(u64, Option<u64>)>() as u64;
   65 +    const CTRL_BYTES_PER_BUCKET: u64 = 1;
   66 +    const GROUP_WIDTH: u64 = 16;
   67 +    let buckets = if entries < 4 {
   68 +        4
   69 +    } else if entries < 8 {
   70 +        8
   71 +    } else {
   72 +        entries
   73 +            .saturating_mul(8)
   74 +            .div_ceil(7)
   75 +            .checked_next_power_of_two()
   76 +            .unwrap_or(u64::MAX)
   77 +    };
   78 +    buckets
   79 +        .saturating_mul(SLOT_BYTES + CTRL_BYTES_PER_BUCKET)
   80 +        .saturating_add(GROUP_WIDTH)
   81 +}

  149 +/// The complete in-memory mapping an index whose `remap` predates batch
  150 +/// translation needs: every address of every fragment in `stored_fragments`
  151 +/// translated through `remapper`, with a row the translator drops mapped
  152 +/// explicitly to `None`. The legacy `remap` treats an address missing from
  153 +/// its map as unchanged, so the fragments listed must be every fragment the
  154 +/// index's files may hold addresses for, retired ones included; a fragment

  156 +/// its stale addresses. An address the translator leaves as it is has the
  157 +/// same meaning either way and is not stored.

  160 +/// counts (from [`BatchRowIdRemapper::fragment_physical_rows`]) put an upper
  161 +/// bound on the map, sized as the hash table really allocates it (see
  162 +/// `hash_map_bytes`), and that plus the translation buffers must fit
  163 +/// [`BatchRowIdRemapper::materialization_budget_bytes`]. The fallback
  164 +/// declines with a [`RemapUnavailable`] when the fragments are unknown, a
  165 +/// fragment cannot be sized or the estimate is over budget; translation
  166 +/// errors propagate as they are.
  167 +pub async fn materialize_remap(
  168 +    remapper: &dyn BatchRowIdRemapper,
  169 +    stored_fragments: Option<&RoaringBitmap>,
  170 +) -> Result<RowAddrRemap> {
  171 +    let Some(fragments) = stored_fragments else {
  172 +        return Err(RemapUnavailable::StoredFragmentsUnknown.into_error());
  173 +    };
  174 +    let mut sized = Vec::with_capacity(fragments.len() as usize);
  175 +    let mut total_rows: u64 = 0;
  176 +    for fragment in fragments.iter() {
  177 +        let rows = remapper
  178 +            .fragment_physical_rows(fragment)
  179 +            .ok_or_else(|| RemapUnavailable::FragmentRowsUnknown { fragment }.into_error())?;
  180 +        total_rows = total_rows.saturating_add(rows);
  181 +        sized.push((fragment, rows));
  182 +    }
  183 +    let budget_bytes = remapper.materialization_budget_bytes();
  184 +    let estimated_bytes = hash_map_bytes(total_rows).saturating_add((BATCH_SIZE as u64) * (8 + 16));
  185 +    if estimated_bytes > budget_bytes {
  186 +        return Err(RemapUnavailable::OverBudget {
  187 +            estimated_bytes,
  188 +            budget_bytes,
  189 +        }
  190 +        .into_error());
  191 +    }
  192 +    let mut map: HashMap<u64, Option<u64>> =
  193 +        HashMap::with_capacity(usize::try_from(total_rows).unwrap_or(usize::MAX));
  194 +    let mut slice: Vec<u64> = Vec::with_capacity(BATCH_SIZE.min(total_rows as usize));
  195 +    for (fragment, rows) in sized {
  196 +        let base = u64::from(fragment) << 32;
  197 +        let mut start = 0u64;
  198 +        while start < rows {
  199 +            let end = rows.min(start + BATCH_SIZE as u64);
  200 +            slice.clear();
  201 +            slice.extend((start..end).map(|offset| base | offset));
  202 +            let translated = remapper.remap_row_ids(&slice).await?;
  203 +            if translated.len() != slice.len() {
  204 +                return Err(Error::internal(format!(
  205 +                    "row-ID remapper returned {} results for {} inputs",
  206 +                    translated.len(),
  207 +                    slice.len()
  208 +                )));
  209 +            }
  210 +            for (address, result) in slice.iter().zip(translated) {
  211 +                if result != Some(*address) {
  212 +                    map.insert(*address, result);
  213 +                }
  214 +            }
  215 +            start = end;
  216 +        }
  217 +    }
  218 +    Ok(RowAddrRemap::direct(map))
  219  }

  362 +            }
  363 +        }
