 5330 +        if let std::collections::hash_map::Entry::Vacant(entry) = bases.entry(id) {
 5331 +            let root = dataset.blob_base_path(id)?;
 5332 +            let store = dataset.object_store(id).await?;
 5333 +            entry.insert((store.store_prefix == owner.object_store.store_prefix, root));
 5334 +        }
 5335 +        let (same_store, root) = bases
 5336 +            .get(&id)
 5337 +            .ok_or_else(|| Error::internal("Missing resolved Managed base"))?;
 5338 +        if *same_store {
 5339 +            let path = join_base_and_relative_path(root, &uri)?;
 5340 +            if path.prefix_match(&owner.base).is_some() {
 5341 +                paths.insert(path);
 5342 +            }
 5343 +        }
 5344 +    }
 5345 +    Ok(paths)
 5346 +}

 5348 +/// Preserve Managed references and adopt Packed/Dedicated objects in place.
 5349 +/// Only Inline payloads are copied. Implicit source bases become explicit when
 5350 +/// a shallow clone rewrites descriptors into its own table base.
 5351 +pub(super) async fn preserve_managed_descriptors(
 5352 +    dataset: &Arc<Dataset>,
 5353 +    field_id: u32,
 5354 +    values: &StructArray,
 5355 +    row_addrs: &[u64],
 5356 +    field: &ArrowField,
 5357 +) -> Result<ArrayRef> {
 5358 +    if values.len() != row_addrs.len() {
 5359 +        return Err(Error::internal(
 5360 +            "Blob descriptors and row addresses have different lengths",
 5361 +        ));
 5362 +    }
 5363 +    let columns = BlobV2DescriptorColumns::new(values);
 5364 +    let mut builder = BlobDescriptorArrayBuilder::new_with_metadata(
 5365 +        field.name(),
 5366 +        field.is_nullable(),
 5367 +        field.metadata().clone(),
 5368 +    );
 5369 +    let mut context = BlobV2ReadContext::new(dataset, field_id);
 5370 +    for (row, row_addr) in row_addrs.iter().enumerate() {
 5371 +        if columns.is_null_blob(row) {
 5372 +            builder.push_null()?;
 5373 +            continue;
 5374 +        }
 5375 +        let kind = BlobKind::try_from(columns.kinds.value(row))?;
 5376 +        match kind {
 5377 +            BlobKind::Managed | BlobKind::ManagedWithBase => {
 5378 +                context.collect_managed(&columns, row, *row_addr).await?;
 5379 +                builder.push(BlobDescriptor::Managed {
 5380 +                    base_id: resolve_managed_base_id(
 5381 +                        dataset,
 5382 +                        field_id,
 5383 +                        *row_addr,
 5384 +                        columns.managed_base_id(row),
 5385 +                    )?,
 5386 +                    uri: columns.blob_uris.value(row).to_string(),
 5387 +                    offset: columns.positions.value(row),
 5388 +                    size: columns.sizes.value(row),
 5389 +                })?;
 5390 +            }
 5391 +            BlobKind::Packed | BlobKind::Dedicated => {
 5392 +                let location = context.blob_read_location(*row_addr).await?;
 5393 +                let root = dataset.blob_base_path(location.base_id)?;
 5394 +                let path = blob_path(
 5395 +                    &location.data_file_dir,
 5396 +                    &location.data_file_key,
 5397 +                    columns.blob_ids.value(row),
 5398 +                );
 5399 +                let uri = path
 5400 +                    .prefix_match(&root)
 5401 +                    .ok_or_else(|| {
 5402 +                        Error::internal(format!("Blob source {path} is outside base {root}"))
 5403 +                    })?
 5404 +                    .map(|part| part.as_ref().to_string())
 5405 +                    .collect::<Vec<_>>()
 5406 +                    .join("/");
 5407 +                builder.push(BlobDescriptor::Managed {
 5408 +                    base_id: location.base_id,
 5409 +                    uri,
 5410 +                    offset: if kind == BlobKind::Dedicated {
 5411 +                        0
 5412 +                    } else {
 5413 +                        columns.positions.value(row)
 5414 +                    },
 5415 +                    size: columns.sizes.value(row),
 5416 +                })?;
 5417 +            }
 5418 +            BlobKind::External => builder.push(BlobDescriptor::External {
 5419 +                base_id: columns.blob_ids.value(row),
 5420 +                uri: columns.blob_uris.value(row).to_string(),
 5421 +                offset: columns.positions.value(row),
 5422 +                size: columns.sizes.value(row),
 5423 +            })?,
 5424 +            BlobKind::Inline => {
 5425 +                let file = context.collect_inline(&columns, row, *row_addr).await?;
 5426 +                builder.push_inline(file.read().await?)?;
 5427 +            }
 5428 +        }
 5429 +    }
 5430 +    Ok(builder.finish()?.into_parts().1)
 5431 +}
