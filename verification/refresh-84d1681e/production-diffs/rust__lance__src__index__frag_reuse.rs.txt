 2322 +/// The copy work a deep clone owes a tagged FRI entry: every file of every
 2323 +/// `_fri/<map_id>/` row map the history references, resolved through the
 2324 +/// manifest's `base_paths` (a chained shallow-then-deep clone finds maps
 2325 +/// living in ancestor bases), as `(relative_path, base_root)` pairs for
 2326 +/// `deep_clone`'s copy loop. Map ids are UUIDs minted at rewrite time, so
 2327 +/// the copies land in the target's fresh `_fri/` namespace collision-free.
 2328 +///
 2329 +/// A history this writer cannot fully interpret (an unknown mapping kind or
 2330 +/// an unknown envelope record) is refused with `NotSupported` before
 2331 +/// anything is copied: an unknown record may reference row maps this writer
 2332 +/// cannot enumerate, and copying an incomplete set would fabricate a
 2333 +/// silently broken "independent" table. This is the same obligation the
 2334 +/// shallow-clone relocation honors, with the same error.
 2335 +pub(crate) async fn collect_tagged_row_map_paths(
 2336 +    dataset: &Dataset,
 2337 +    entry: &IndexMetadata,
 2338 +) -> lance_core::Result<Vec<(String, Path)>> {
 2339 +    use futures::TryStreamExt;
 2340 +    use lance_index::frag_reuse::stable_partition::MAPPING_FILE;
 2341 +    use lance_table::system_index::frag_reuse::ledger::Mapping;
 2342 +
 2343 +    let ledger = decode_frag_reuse_ledger(dataset, entry).await?;
 2344 +    if ledger.has_unsupported_transitions() {
 2345 +        return Err(Error::not_supported(
 2346 +            "the fragment reuse history contains mappings this writer cannot relocate; \
 2347 +             upgrade to a newer version of Lance before cloning this table",
 2348 +        ));
 2349 +    }
 2350 +
 2351 +    let mut seen: HashSet<(Option<u32>, &str)> = HashSet::new();
 2352 +    let mut paths = Vec::new();
 2353 +    for transition in ledger.transitions() {
 2354 +        let Mapping::StablePartition(reference) = transition.mapping() else {
 2355 +            continue;
 2356 +        };
 2357 +        if !seen.insert((reference.base_id, reference.map_id.as_str())) {
 2358 +            continue;
 2359 +        }
 2360 +        // The same base-aware resolution the reader opens the map with.
 2361 +        let base_root = match reference.base_id {
 2362 +            None => dataset.base.clone(),
 2363 +            Some(id) => dataset
 2364 +                .manifest
 2365 +                .base_paths
 2366 +                .get(&id)
 2367 +                .ok_or_else(|| {
 2368 +                    Error::invalid_input(format!(
 2369 +                        "mapping {} references missing base {id}",
 2370 +                        reference.map_id
 2371 +                    ))
 2372 +                })?
 2373 +                .extract_path(dataset.session.store_registry())?,
 2374 +        };
 2375 +        let map_dir = base_root
 2376 +            .clone()
 2377 +            .join("_fri")
 2378 +            .join(reference.map_id.as_str());
 2379 +        let mut mapping_size = None;
 2380 +        let mut stream = dataset.object_store.read_dir_all(&map_dir, None);
 2381 +        loop {
 2382 +            match stream.try_next().await {
 2383 +                Ok(Some(meta)) => {
 2384 +                    let relative = meta
 2385 +                        .location
 2386 +                        .as_ref()
 2387 +                        .strip_prefix(map_dir.as_ref())
 2388 +                        .ok_or_else(|| {
 2389 +                            Error::internal(format!(
 2390 +                                "listing {} returned {} outside it",
 2391 +                                map_dir, meta.location
 2392 +                            ))
 2393 +                        })?
 2394 +                        .trim_start_matches('/');
 2395 +                    if relative == MAPPING_FILE {
 2396 +                        mapping_size = Some(meta.size);
 2397 +                    }
 2398 +                    paths.push((
 2399 +                        format!("_fri/{}/{relative}", reference.map_id),
 2400 +                        base_root.clone(),
 2401 +                    ));
 2402 +                }
 2403 +                Ok(None) => break,
 2404 +                Err(Error::NotFound { .. }) => break,
 2405 +                Err(error) => return Err(error),
 2406 +            }
 2407 +        }
 2408 +        // The copy must produce a working clone: the map's mapping file has
 2409 +        // to be there with the size the history records for it (the reader
 2410 +        // opens the file by that size). Refuse the clone now rather than
 2411 +        // hand the target a history whose first translating query fails.
 2412 +        match mapping_size {
 2413 +            Some(size) if size == reference.map_size_bytes => {}
 2414 +            Some(size) => {
 2415 +                return Err(Error::corrupt_file_named(
 2416 +                    "FRI details",
 2417 +                    format!(
 2418 +                        "row map {} has a {MAPPING_FILE} of {size} bytes under its base, \
 2419 +                         the history records {} bytes",
 2420 +                        reference.map_id, reference.map_size_bytes
 2421 +                    ),
 2422 +                ));
 2423 +            }
 2424 +            None => {
 2425 +                return Err(Error::corrupt_file_named(
 2426 +                    "FRI details",
 2427 +                    format!(
 2428 +                        "row map {} has no {MAPPING_FILE} under its base",
 2429 +                        reference.map_id
 2430 +                    ),
 2431 +                ));
 2432 +            }
 2433 +        }
 2434 +    }
 2435 +    Ok(paths)
 2436 +}
