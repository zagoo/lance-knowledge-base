  581 +fn split_trim_elements(content: &[u8]) -> Result<Vec<TrimElement>> {
  582 +    use bytes::Buf;
  583 +    use prost::encoding::{WireType, decode_key, decode_varint};
  584 +
  585 +    let corrupt = |message: String| Error::corrupt_file_named("FRI details", message);
  586 +    let total = content.len();
  587 +    let mut buf = bytes::Bytes::copy_from_slice(content);
  588 +    let mut elements = Vec::new();
  589 +    while buf.has_remaining() {
  590 +        let start = total - buf.remaining();
  591 +        let (tag, wire_type) = decode_key(&mut buf).map_err(|e| corrupt(e.to_string()))?;
  592 +        // Fields 1 and 2 are the only known records and both are messages. An
  593 +        // unknown record may participate in lineage in ways this client cannot
  594 +        // reason about, so refuse to trim rather than silently dropping or
  595 +        // blindly retaining it.
  596 +        if wire_type != WireType::LengthDelimited || !matches!(tag, 1 | 2) {
  597 +            return Err(Error::not_supported(format!(
  598 +                "the tagged FRI history carries an unknown record (field {tag}); \
  599 +                 upgrade to a newer version of Lance before trimming"
  600 +            )));
  601 +        }
  602 +        let length = decode_varint(&mut buf).map_err(|e| corrupt(e.to_string()))?;
  603 +        if length > buf.remaining() as u64 {
  604 +            return Err(corrupt(format!(
  605 +                "field {tag} length {length} exceeds remaining {} bytes",
  606 +                buf.remaining()
  607 +            )));
  608 +        }
  609 +        let payload = buf.split_to(length as usize);
  610 +        let end = total - buf.remaining();
  611 +        let record = match tag {
  612 +            1 => TrimRecord::LegacyVersion(
  613 +                pb_fri::Version::decode(payload)
  614 +                    .map_err(|e| corrupt(e.to_string()))?
  615 +                    .try_into()?,
  616 +            ),
  617 +            _ => TrimRecord::Transition(
  618 +                pb_fri::Transition::decode(payload).map_err(|e| corrupt(e.to_string()))?,
  619 +            ),
  620 +        };
  621 +        elements.push(TrimElement {
  622 +            raw: content[start..end].to_vec(),
  623 +            record,
  624 +        });
  625 +    }
  626 +    Ok(elements)
  627 +}

  713 +    }
  714 +
  715 +    let mut retained = vec![false; elements.len()];
  716 +    for (position, element) in elements.iter().enumerate() {
  717 +        match &element.record {
  718 +            TrimRecord::LegacyVersion(version) => {
  719 +                let check_results = indices
  720 +                    .iter()
  721 +                    .map(|idx| is_index_remap_caught_up(version, idx, &chain_frag_bitmap))
  722 +                    .collect::<Vec<_>>();
  723 +                if check_results
  724 +                    .iter()
  725 +                    .any(|r| matches!(r, Err(Error::InvalidInput { .. })))
  726 +                {
  727 +                    // If the check fails, the reuse version is likely corrupted,
  728 +                    // do not retain it (v0 behavior).
  729 +                    continue;
  730 +                }
  731 +                if !check_results.into_iter().all(|r| r.unwrap()) {
  732 +                    retained[position] = true;
  733 +                }
  734 +            }
  735 +            TrimRecord::Transition(transition) => {
  736 +                retained[position] = is_transition_needed(transition, &index_groups);
  737 +            }
  738 +        }
  739 +    }
  740 +
  741 +    // Transitive chain retention (forward closure over producer->consumer
  742 +    // edges).
  743 +    struct Node {
  744 +        element: usize,
  745 +        sources: Vec<u64>,
  746 +        destinations: Vec<u64>,
  747 +    }
  748 +    let mut nodes = Vec::new();
  749 +    for (position, element) in elements.iter().enumerate() {
  750 +        match &element.record {
  751 +            TrimRecord::LegacyVersion(version) => {
  752 +                for group in version.groups.iter() {
  753 +                    nodes.push(Node {
  754 +                        element: position,
  755 +                        sources: group.old_frags.iter().map(|f| f.id).collect(),
  756 +                        destinations: group.new_frags.iter().map(|f| f.id).collect(),
  757 +                    });
  758 +                }
  759 +            }
  760 +            TrimRecord::Transition(transition) => nodes.push(Node {
  761 +                element: position,
  762 +                sources: transition.sources.iter().map(|d| d.id).collect(),
  763 +                destinations: transition.destinations.iter().map(|d| d.id).collect(),
  764 +            }),
  765 +        }
  766 +    }
  767 +    let mut consumer_of: HashMap<u64, usize> = HashMap::new();
  768 +    let mut element_nodes: HashMap<usize, Vec<usize>> = HashMap::new();
  769 +    for (node_index, node) in nodes.iter().enumerate() {
  770 +        for source in &node.sources {
  771 +            consumer_of.insert(*source, node_index);
  772 +        }
  773 +        element_nodes
  774 +            .entry(node.element)
  775 +            .or_default()
  776 +            .push(node_index);
  777 +    }
  778 +    let mut node_retained = vec![false; nodes.len()];
  779 +    let mut queue: VecDeque<usize> = VecDeque::new();
  780 +    for (node_index, node) in nodes.iter().enumerate() {
  781 +        if retained[node.element] {
  782 +            node_retained[node_index] = true;
  783 +            queue.push_back(node_index);
  784 +        }
  785 +    }
  786 +    while let Some(node_index) = queue.pop_front() {
  787 +        for destination in nodes[node_index].destinations.clone() {
  788 +            let Some(&consumer) = consumer_of.get(&destination) else {
  789 +                continue;
  790 +            };
  791 +            if node_retained[consumer] {
  792 +                continue;
  793 +            }
  794 +            node_retained[consumer] = true;
  795 +            queue.push_back(consumer);
  796 +            let element = nodes[consumer].element;
  797 +            if !retained[element] {
  798 +                retained[element] = true;
  799 +                for &sibling in element_nodes[&element].iter() {
  800 +                    if !node_retained[sibling] {
  801 +                        node_retained[sibling] = true;
  802 +                        queue.push_back(sibling);
  803 +                    }
  804 +                }
  805 +            }
  806 +        }
  807 +    }
  808 +
  809 +    retained
  810 +}

  812 +/// Derive the trimmed tagged entry from the CURRENT manifest state.
  813 +///
  814 +/// Zero payload IO: retention is computed from the entry's digests and the
  815 +/// stored index metadata alone. Row-map files released by trimming a
  816 +/// stable-partition transition are collected separately by dataset cleanup.
  817 +pub(crate) async fn derive_tagged_trim(dataset: &Dataset) -> lance_core::Result<TaggedTrimOutcome> {
  818 +    let indices = read_manifest_indexes(
  819 +        &dataset.object_store,
  820 +        &dataset.manifest_location,
  821 +        &dataset.manifest,
  822 +    )
  823 +    .await?;
  824 +    let Some(entry) = indices
  825 +        .iter()
  826 +        .find(|idx| idx.name == FRAG_REUSE_INDEX_NAME)
  827 +        .cloned()
  828 +    else {
  829 +        return Ok(TaggedTrimOutcome::NothingToTrim);
  830 +    };
  831 +    if entry.index_version == 0 {
  832 +        // Reachable only through a rebase whose entry was concurrently
  833 +        // replaced by a v0 one; nothing tagged remains to trim.
  834 +        return Ok(TaggedTrimOutcome::NothingToTrim);
  835 +    }
  836 +
  837 +    let content = load_raw_frag_reuse_content(dataset, &entry).await?;
  838 +    // Full ledger validation (lineage order, digest conservation, mapping
  839 +    // presence) plus unsupported-transition detection before interpreting
  840 +    // anything.
  841 +    let ledger = decode_frag_reuse_ledger_from_content(entry.index_version, &content).await?;
  842 +    if ledger.has_unsupported_transitions() {
  843 +        return Err(Error::not_supported(
  844 +            "the tagged FRI history carries transitions this client cannot interpret; \
  845 +             upgrade to a newer version of Lance before trimming",
  846 +        ));
  847 +    }
  848 +    let elements = split_trim_elements(&content)?;
  849 +    let retained = compute_tagged_retention(&elements, &indices);
  850 +
  851 +    if retained.iter().all(|kept| *kept) {
  852 +        return Ok(TaggedTrimOutcome::NothingToTrim);
  853 +    }
  854 +    if retained.iter().all(|kept| !*kept) {
  855 +        return Ok(TaggedTrimOutcome::Delete {
  856 +            current_entry: entry,
  857 +        });
  858 +    }
  859 +
  860 +    let mut new_content = Vec::new();
  861 +    let mut fragment_bitmap = RoaringBitmap::new();
  862 +    for (element, kept) in elements.iter().zip(retained.iter()) {
  863 +        if !kept {
  864 +            continue;
  865 +        }
  866 +        new_content.extend_from_slice(&element.raw);
  867 +        match &element.record {
  868 +            TrimRecord::LegacyVersion(version) => {
  869 +                fragment_bitmap.extend(version.old_frag_ids().iter().map(|&id| id as u32));
  870 +                fragment_bitmap.extend(version.new_frag_ids().iter().map(|&id| id as u32));
  871 +            }
  872 +            TrimRecord::Transition(transition) => {
  873 +                for digest in transition
  874 +                    .sources
  875 +                    .iter()
  876 +                    .chain(transition.destinations.iter())
  877 +                {
  878 +                    fragment_bitmap.insert(digest.id as u32);
  879 +                }
  880 +            }
  881 +        }
  882 +    }
  883 +    // The filtered history must itself decode as a well-formed ledger.
  884 +    decode_frag_reuse_ledger_from_content(entry.index_version, &new_content).await?;
  885 +    let new_entry = build_tagged_frag_reuse_entry(dataset, new_content, fragment_bitmap).await?;
  886 +    Ok(TaggedTrimOutcome::Replace {
  887 +        new_entry,
  888 +        current_entry: entry,
  889 +    })
  890 +}
