  142 +fn canonicalize_merge_replacements(
  143 +    manifest: &Manifest,
  144 +    schema: &mut Schema,
  145 +    fragments: &mut [Fragment],
  146 +    retained_files: &HashSet<DataFileIdentity>,
  147 +    read_schema: &Schema,
  148 +) -> Result<()> {
  149 +    let mut replaced_field_ids = HashSet::new();
  150 +    for fragment in fragments.iter() {
  151 +        let retained_field_ids = fragment
  152 +            .referenced_lance_files()
  153 +            .filter(|file| retained_files.contains(&(file.base_id, file.path.clone())))
  154 +            .flat_map(|file| file.fields.iter().copied())
  155 +            .collect::<HashSet<_>>();
  156 +        replaced_field_ids.extend(
  157 +            fragment
  158 +                .referenced_lance_files()
  159 +                .filter(|file| !retained_files.contains(&(file.base_id, file.path.clone())))
  160 +                .flat_map(|file| file.fields.iter().copied())
  161 +                .filter(|field_id| retained_field_ids.contains(field_id)),
  162 +        );
  163 +    }
  164 +    let original = schema.clone();
  165 +    for field in &mut schema.fields {
  166 +        clear_replaced_and_new_field_ids(field, read_schema, &replaced_field_ids);
  167 +    }
  168 +    schema.try_set_field_id(Some(manifest.max_field_id()))?;
  169 +    schema.validate()?;
  170 +    schema.verify_primary_key()?;
  171 +
  172 +    let field_id_remap = original
  173 +        .fields_pre_order()
  174 +        .zip(schema.fields_pre_order())
  175 +        .filter(|(original, _)| original.id >= 0)
  176 +        .map(|(original, canonical)| (original.id, canonical.id))
  177 +        .collect();
  178 +    remap_fragment_field_ids(fragments, &field_id_remap, retained_files);
  179 +    Ok(())
  180 +}

  245 +fn canonicalize_field(
  246 +    field: &mut Field,
  247 +    parent_id: i32,
  248 +    base_schema: &Schema,
  249 +    base_parent: Option<&Field>,
  250 +    identity_schema: Option<&Schema>,
  251 +) -> bool {
  252 +    let same_name = match base_parent {
  253 +        Some(parent) => parent.children.iter().find(|base| base.name == field.name),

  257 +            .find(|base| base.name == field.name),
  258 +    };
  259 +    let allow_id_binding =
  260 +        identity_schema.is_some_and(|schema| schema.field_by_id(field.id).is_some());
  261 +    let by_id = (allow_id_binding && field.id >= 0)
  262 +        .then(|| base_schema.field_by_id(field.id))
  263 +        .flatten()
  264 +        .filter(|base| base.parent_id == parent_id);
  265 +    let base_field = if allow_id_binding && field.id >= 0 {
  266 +        by_id
  267 +            .filter(|base| base.logical_type == field.logical_type)
  268 +            .or_else(|| same_name.filter(|base| base.logical_type == field.logical_type))
  269 +    } else {
  270 +        same_name.filter(|base| base.logical_type == field.logical_type)
  271 +    };
  272 +
  273 +    let Some(base_field) = base_field else {
  274 +        clear_field_ids(field);
  275 +        return false;
  276 +    };
  277 +
  278 +    field.id = base_field.id;
  279 +    field.parent_id = parent_id;
  280 +    let mut all_children_match = true;
  281 +    for child in &mut field.children {
  282 +        if !canonicalize_field(
  283 +            child,
  284 +            field.id,
  285 +            base_schema,
  286 +            Some(base_field),
  287 +            identity_schema,

  289 +            all_children_match = false;
  290 +        }
  291 +    }
  292 +    all_children_match

  303 +fn resolve_fragment_field_ids(
  304 +    fragments: &mut [Fragment],
  305 +    field_id_remap: &HashMap<i32, i32>,
  306 +    retained_files: &HashSet<DataFileIdentity>,
  307 +    schema: &Schema,
  308 +) -> Result<()> {
  309 +    let canonical_ids = schema.field_ids();
  310 +    for fragment in fragments {
  311 +        // Raw Arrow overwrite fragments may have been written either by a
  312 +        // standalone writer using the source schema IDs or by a
  313 +        // dataset-aware writer using canonical IDs. Resolve that namespace
  314 +        // once for the whole fragment so split files cannot disagree. If
  315 +        // both interpretations are possible and produce different
  316 +        // identities then there is no safe mapping without provenance.
  317 +        let fragment_field_ids = fragment
  318 +            .referenced_lance_files()
  319 +            .filter(|file| !retained_files.contains(&(file.base_id, file.path.clone())))
  320 +            .flat_map(|file| file.fields.iter().copied())
  321 +            .filter(|field_id| *field_id >= 0)
  322 +            .collect::<HashSet<_>>();
  323 +        let canonical_source = fragment_field_ids
  324 +            .iter()
  325 +            .all(|field_id| canonical_ids.contains(field_id));
  326 +        let raw_source = fragment_field_ids
  327 +            .iter()
  328 +            .all(|field_id| field_id_remap.contains_key(field_id));
  329 +        let raw_changes_identity = fragment_field_ids
  330 +            .iter()
  331 +            .any(|field_id| field_id_remap.get(field_id) != Some(field_id));
  332 +
  333 +        match (canonical_source, raw_source, raw_changes_identity) {
  334 +            (true, true, true) => {
  335 +                return Err(Error::invalid_input(format!(
  336 +                    "Fragment {} has ambiguous raw Arrow field IDs; its file mappings can be interpreted as either source or canonical identities",
  337 +                    fragment.id
  338 +                )));
  339 +            }
  340 +            (true, _, _) => continue,
  341 +            (false, true, _) => {}
  342 +            (false, false, _) => {
  343 +                return Err(Error::invalid_input(format!(
  344 +                    "Fragment {} field IDs do not match either the raw Arrow source schema or the canonical replacement schema",
  345 +                    fragment.id
  346 +                )));
  347 +            }
  348 +        }
  349 +        remap_fragment_field_ids(
  350 +            std::slice::from_mut(fragment),
  351 +            field_id_remap,
  352 +            retained_files,
  353 +        );
  354 +    }
  355 +    Ok(())
  356 +}

  358 +fn remap_fragment_field_ids(
  359 +    fragments: &mut [Fragment],
  360 +    field_id_remap: &HashMap<i32, i32>,
  361 +    retained_files: &HashSet<DataFileIdentity>,
  362 +) {

  372 +            }
  373 +        }
  374 +    }
