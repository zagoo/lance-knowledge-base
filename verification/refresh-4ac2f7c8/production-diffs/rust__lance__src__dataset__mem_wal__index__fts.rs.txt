 5188 +    fn create(&self, ctx: &MemIndexBuildContext<'_>) -> Result<Arc<dyn MemIndex>> {
 5189 +        let (column, field_id) = ctx.single_column()?;
 5190 +        let params = ctx.params::<FtsParams>()?;
 5191 +        let index = match &params.resolved_field {
 5192 +            Some(resolved) => FtsMemIndex::try_with_resolved_field(
 5193 +                field_id,
 5194 +                column.to_string(),
 5195 +                params.params.clone(),
 5196 +                resolved.clone(),
 5197 +            )?,
 5198 +            None => {
 5199 +                FtsMemIndex::try_with_params(field_id, column.to_string(), params.params.clone())?
 5200 +            }
 5201 +        };
 5202 +        Ok(Arc::new(index))
 5203 +    }
 5204 +}

 5206 +impl FtsMemIndexPlugin {
 5207 +    /// What a full-text index covers and needs, read from the base-table index
 5208 +    /// entry and the schema alone.
 5209 +    pub(crate) fn resolve_from_metadata(
 5210 +        name: &str,
 5211 +        schema: &LanceSchema,
 5212 +        index_meta: &IndexMetadata,
 5213 +    ) -> Result<ResolvedIndex> {
 5214 +        let [field_id] = index_meta.fields.as_slice() else {
 5215 +            return Err(Error::invalid_input(format!(
 5216 +                "index '{name}' covers {} columns, but this kind covers exactly one",
 5217 +                index_meta.fields.len()
 5218 +            )));
 5219 +        };
 5220 +
 5221 +        let details = match &index_meta.index_details {
 5222 +            Some(details) => pbold::InvertedIndexDetails::decode(details.value.as_slice())
 5223 +                .map_err(|err| {
 5224 +                    Error::io(format!(
 5225 +                        "failed to decode InvertedIndexDetails for MemWAL FTS index \
 5226 +                         '{name}': {err}"
 5227 +                    ))
 5228 +                })?,
 5229 +            None => pbold::InvertedIndexDetails::default(),
 5230 +        };
 5231 +        let details =
 5232 +            crate::index::scalar::inverted::normalize_inverted_details(index_meta, details)?;
 5233 +        let params = InvertedIndexParams::try_from(&details)?;
 5234 +
 5235 +        // The index covers the resolved path, the name a query uses.
 5236 +        let resolved = crate::index::scalar::inverted::resolve_fts_field_by_id(
 5237 +            schema,
 5238 +            *field_id,
 5239 +            params.get_document_granularity(),
 5240 +        )?;
 5241 +
 5242 +        let field_id = resolved.final_field_id;
 5243 +        Ok(ResolvedIndex::with_params(
 5244 +            vec![resolved.canonical_path.clone()],
 5245 +            FtsParams {
 5246 +                params,
 5247 +                resolved_field: Some(resolved),
 5248 +            },
 5249 +        )
 5250 +        .with_field_ids(vec![field_id]))
 5251 +    }
 5252 +}

 5254 +/// What an in-memory full-text index needs before it takes a row.
 5255 +#[derive(Debug, Clone)]
 5256 +pub struct FtsParams {
 5257 +    /// Tokenizer, granularity and format, from the base-table index.
 5258 +    pub params: InvertedIndexParams,
 5259 +    /// The path through nested types the index covers; `None` resolves it
 5260 +    /// from the first batch.
 5261 +    pub(crate) resolved_field: Option<ResolvedFtsField>,
 5262 +}

 5266 +    fn eq(&self, other: &Self) -> bool {
 5267 +        self.params == other.params
 5268 +    }
 5269 +}
