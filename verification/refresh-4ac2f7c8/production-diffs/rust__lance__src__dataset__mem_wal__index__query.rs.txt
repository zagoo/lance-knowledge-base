    4 +//! What a memtable index can be asked, and what it answers.
    5 +//!
    6 +//! [`MemQuery`] is open to any query type, Lance's scalar index queries
    7 +//! included. [`MemMatches`] is either a filter's set of rows or a search's
    8 +//! ranked list.

   21 +/// A question put to a memtable index.
   22 +pub trait MemQuery: Debug + Send + Sync {
   23 +    /// The concrete query, for an index to recognise the ones it answers.
   24 +    fn as_any(&self) -> &dyn Any;
   25 +}
   26 +
   27 +impl<T: AnyQuery> MemQuery for T {
   28 +    fn as_any(&self) -> &dyn Any {
   29 +        AnyQuery::as_any(self)
   30 +    }
   31 +}

   33 +/// A Lance scalar query behind a trait object, as the expression pass hands it

   38 +impl MemQuery for ScalarQuery<'_> {
   39 +    fn as_any(&self) -> &dyn Any {
   40 +        self.0.as_any()
   41 +    }

   44 +/// A set of positions in one memtable.
   45 +#[derive(Debug, Clone, Default, PartialEq)]
   46 +pub struct PositionSet(RoaringTreemap);
   47 +
   48 +impl PositionSet {
   49 +    /// The empty set.
   50 +    pub fn empty() -> Self {
   51 +        Self(RoaringTreemap::new())
   52 +    }
   53 +
   54 +    /// Every position up to and including `max_visible`.
   55 +    pub fn all_visible(max_visible: RowPosition) -> Self {
   56 +        let mut positions = RoaringTreemap::new();
   57 +        positions.insert_range(0..=max_visible);
   58 +        Self(positions)
   59 +    }
   60 +
   61 +    /// Whether the set holds no position.
   62 +    pub fn is_empty(&self) -> bool {
   63 +        self.0.is_empty()
   64 +    }
   65 +
   66 +    /// How many positions the set holds.
   67 +    pub fn len(&self) -> u64 {
   68 +        self.0.len()
   69 +    }
   70 +
   71 +    /// Whether `position` is in the set.
   72 +    pub fn contains(&self, position: RowPosition) -> bool {
   73 +        self.0.contains(position)
   74 +    }
   75 +
   76 +    /// Add one position.
   77 +    pub fn insert(&mut self, position: RowPosition) {
   78 +        self.0.insert(position);
   79 +    }
   80 +
   81 +    /// The positions in ascending order.
   82 +    pub fn iter(&self) -> impl Iterator<Item = RowPosition> + '_ {
   83 +        self.0.iter()
   84 +    }
   85 +
   86 +    /// Drop everything above `max_visible`.
   87 +    pub fn truncate_to(mut self, max_visible: RowPosition) -> Self {
   88 +        if let Some(first_hidden) = max_visible.checked_add(1) {
   89 +            self.0.remove_range(first_hidden..);
   90 +        }
   91 +        self
   92 +    }
   93 +}
   94 +
   95 +impl FromIterator<RowPosition> for PositionSet {
   96 +    fn from_iter<I: IntoIterator<Item = RowPosition>>(iter: I) -> Self {
   97 +        Self(iter.into_iter().collect())
   98 +    }
   99 +}
  100 +
  101 +impl From<RoaringTreemap> for PositionSet {
  102 +    fn from(map: RoaringTreemap) -> Self {
  103 +        Self(map)
  104 +    }
  105 +}
  106 +
  107 +impl From<PositionSet> for Vec<RowPosition> {
  108 +    fn from(set: PositionSet) -> Self {
  109 +        set.0.into_iter().collect()
  110 +    }
  111 +}
  112 +
  113 +impl std::ops::BitAnd for PositionSet {
  114 +    type Output = Self;
  115 +    fn bitand(self, rhs: Self) -> Self {
  116 +        Self(self.0 & rhs.0)
  117 +    }
  118 +}
  119 +
  120 +impl std::ops::BitOr for PositionSet {
  121 +    type Output = Self;
  122 +    fn bitor(self, rhs: Self) -> Self {
  123 +        Self(self.0 | rhs.0)
  124 +    }
  125 +}

  127 +/// Which rows a filter matched, as two bounds like Lance's scalar
  128 +/// `SearchResult`: exact when they are equal, otherwise the caller re-checks
  129 +/// the rows in `at_most`.
  130 +#[derive(Debug, Clone, PartialEq)]
  131 +pub struct MemSearchResult {
  132 +    /// Rows that match.
  133 +    pub at_least: PositionSet,
  134 +    /// Rows that may match; nothing outside them does.
  135 +    pub at_most: PositionSet,
  136 +}
  137 +
  138 +impl MemSearchResult {
  139 +    /// A settled answer: exactly these rows match.
  140 +    pub fn exact(positions: PositionSet) -> Self {
  141 +        Self {
  142 +            at_least: positions.clone(),
  143 +            at_most: positions,
  144 +        }
  145 +    }
  146 +
  147 +    /// A narrowed answer: nothing outside `positions` matches, and the caller
  148 +    /// re-checks what is inside.
  149 +    pub fn at_most(positions: PositionSet) -> Self {
  150 +        Self {
  151 +            at_least: PositionSet::empty(),
  152 +            at_most: positions,
  153 +        }
  154 +    }
  155 +
  156 +    /// Nothing matches, and that is settled.
  157 +    pub fn empty() -> Self {
  158 +        Self::exact(PositionSet::empty())
  159 +    }
  160 +
  161 +    /// Whether the answer is settled and needs no re-check.
  162 +    pub fn is_exact(&self) -> bool {
  163 +        self.at_least == self.at_most
  164 +    }
  165 +
  166 +    /// Drop everything above `max_visible` from both endpoints.
  167 +    pub fn truncate_to(self, max_visible: RowPosition) -> Self {
  168 +        Self {
  169 +            at_least: self.at_least.truncate_to(max_visible),
  170 +            at_most: self.at_most.truncate_to(max_visible),
  171 +        }
  172 +    }
  173 +}
  174 +
  175 +impl std::ops::BitAnd for MemSearchResult {
  176 +    type Output = Self;
  177 +
  178 +    /// Rows matching both.
  179 +    fn bitand(self, rhs: Self) -> Self {
  180 +        Self {
  181 +            at_least: self.at_least & rhs.at_least,
  182 +            at_most: self.at_most & rhs.at_most,
  183 +        }
  184 +    }
  185 +}
  186 +
  187 +impl std::ops::BitOr for MemSearchResult {
  188 +    type Output = Self;
  189 +
  190 +    /// Rows matching either.
  191 +    fn bitor(self, rhs: Self) -> Self {
  192 +        Self {
  193 +            at_least: self.at_least | rhs.at_least,
  194 +            at_most: self.at_most | rhs.at_most,
  195 +        }
  196 +    }
  197 +}

  199 +/// One row a ranked search returned.
  200 +#[derive(Debug, Clone, PartialEq)]
  201 +pub struct RankedMatch {
  202 +    /// Where the row sits in the memtable.
  203 +    pub position: RowPosition,
  204 +    /// Its score: a distance, lower is better, or a full-text relevance,
  205 +    /// higher is better.
  206 +    pub score: f32,
  207 +    /// The indices of the list element that matched, outermost first, for a
  208 +    /// document nested in lists.
  209 +    pub element: Option<Vec<u32>>,
  210 +}
  211 +
  212 +impl RankedMatch {
  213 +    /// A match on a whole row.
  214 +    pub fn new(position: RowPosition, score: f32) -> Self {
  215 +        Self {
  216 +            position,
  217 +            score,
  218 +            element: None,
  219 +        }
  220 +    }
  221 +
  222 +    /// A match on one element of a nested document.
  223 +    pub fn nested(position: RowPosition, score: f32, element: Vec<u32>) -> Self {
  224 +        Self {
  225 +            position,
  226 +            score,
  227 +            element: Some(element),
  228 +        }
  229 +    }
  230 +}
  231 +
  232 +/// What a search answered.
  233 +#[derive(Debug, Clone, PartialEq)]
  234 +pub enum MemMatches {
  235 +    /// The rows a filter matched.
  236 +    Filter(MemSearchResult),
  237 +    /// The rows a vector or full-text search found, best first.
  238 +    Ranked(Vec<RankedMatch>),
  239 +}
  240 +
  241 +impl MemMatches {
  242 +    /// A settled set of rows.
  243 +    pub fn exact(positions: impl IntoIterator<Item = RowPosition>) -> Self {
  244 +        Self::Filter(MemSearchResult::exact(positions.into_iter().collect()))
  245 +    }
  246 +
  247 +    /// Candidate rows the caller re-checks.
  248 +    pub fn at_most(positions: impl IntoIterator<Item = RowPosition>) -> Self {
  249 +        Self::Filter(MemSearchResult::at_most(positions.into_iter().collect()))
  250 +    }
  251 +
  252 +    /// Rows in rank order.
  253 +    pub fn ranked(matches: Vec<RankedMatch>) -> Self {
  254 +        Self::Ranked(matches)
  255 +    }
  256 +
  257 +    /// The filter result, when that is what was asked for.
  258 +    pub fn as_filter(&self) -> Option<&MemSearchResult> {
  259 +        match self {
  260 +            Self::Filter(result) => Some(result),
  261 +            Self::Ranked(_) => None,
  262 +        }
  263 +    }
  264 +
  265 +    /// The ranked matches, when that is what was asked for.
  266 +    pub fn as_ranked(&self) -> Option<&[RankedMatch]> {
  267 +        match self {
  268 +            Self::Ranked(matches) => Some(matches),
  269 +            Self::Filter(_) => None,
  270 +        }
  271 +    }
  272 +}

  274 +/// What a search needs to know besides the query itself.
  275 +#[derive(Debug, Clone, Copy)]
  276 +pub struct SearchContext {
  277 +    /// The highest position a reader may see. An index may hold rows past it
  278 +    /// and must not return them.
  279 +    pub max_visible: RowPosition,
  280 +    /// How many matching rows are worth listing. An index that would list more
  281 +    /// may decline with `Ok(None)`, and the caller reads every row.
  282 +    pub match_budget: Option<u64>,
  283 +}
  284 +
  285 +impl SearchContext {
  286 +    /// A search over everything visible up to `max_visible`, with no budget.
  287 +    pub fn new(max_visible: RowPosition) -> Self {
  288 +        Self {
  289 +            max_visible,
  290 +            match_budget: None,
  291 +        }
  292 +    }
  293 +
  294 +    /// The same search, declinable once it matches more than `budget` rows.
  295 +    pub fn with_match_budget(mut self, budget: u64) -> Self {
  296 +        self.match_budget = Some(budget);
  297 +        self
  298 +    }
  299 +}
