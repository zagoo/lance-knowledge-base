   23 +/// Represents a valid JSON Path.
   24 +#[derive(Debug, Clone, PartialEq)]
   25 +pub enum Path<'a> {
   26 +    /// `$` represents the root node or element.
   27 +    Root,
   28 +    /// `@` represents the current node or element being processed in the filter expression.
   29 +    Current,
   30 +    /// `.*` represents selecting all elements in an Object.
   31 +    DotWildcard,
   32 +    /// `.**` represents recursive selecting all elements in Array and Object.
   33 +    /// The optional `RecursiveLevel` indicates the selected levels.
   34 +    RecursiveDotWildcard(Option<RecursiveLevel>),
   35 +    /// `[*]` represents selecting all elements in an Array.
   36 +    BracketWildcard,
   37 +    /// `.<name>` represents selecting element that matched the name in an Object, like `$.event`.
   38 +    /// The name can also be written as a string literal, allowing the name to contain special characters, like `$." $price"`.
   39 +    DotField(Cow<'a, str>),
   40 +    /// `:<name>` represents selecting element that matched the name in an Object, like `$:event`.
   41 +    ColonField(Cow<'a, str>),
   42 +    /// `["<name>"]` represents selecting element that matched the name in an Object, like `$["event"]`.
   43 +    ObjectField(Cow<'a, str>),
   44 +    /// `[<index1>,<index2>,..]` represents selecting elements specified by the indices in an Array.
   45 +    /// There are several forms of index.
   46 +    /// 1. A single number representing the 0-based `n-th` element in the Array.
   47 +    ///    e.g. `$[0]` represents the first element in an Array.
   48 +    /// 2. The keyword `last` represents the last element in the Array,
   49 +    ///    and last minus a number represents the n-th element before the last element,
   50 +    ///    e.g. `$[last-1]` represents the penultimate element.
   51 +    /// 3. The keyword `to` between two numbers represent all elements of a range in an Array,
   52 +    ///    e.g. `$[1 to last]` represents all the elements in the Array from the second to the last.
   53 +    ///
   54 +    /// There can be more than one index, e.g. `$[0, last-1 to last, 5]` represents the first,
   55 +    /// the last two, and the sixth element in an Array.
   56 +    ArrayIndices(Vec<ArrayIndex>),
   57 +    /// `?(<expression>)` represents selecting all elements in an object or array that match the filter expression, like `$.book[?(@.price < 10)]`.
   58 +    FilterExpr(Box<Expr<'a>>),
   59 +    /// `<expression>` standalone filter expression, like `$.book[*].price > 10`,
   60 +    /// and arithmetic expression, like `-$.a[*]` or `$.a + 3`
   61 +    Expr(Box<Expr<'a>>),
   62 +}

   82 +impl ArrayIndex {
   83 +    /// Converts an `ArrayIndex` to a `HashSet` of indices that should be selected from an Array.
   84 +    ///
   85 +    /// # Arguments
   86 +    ///
   87 +    /// * `length` - The length of the array.
   88 +    ///
   89 +    /// # Returns
   90 +    ///
   91 +    /// A `HashSet<usize>` containing the indices to select.
   92 +    pub(crate) fn to_indices(&self, length: usize) -> HashSet<usize> {
   93 +        let length = length as i32;
   94 +
   95 +        let mut indices = HashSet::new();
   96 +        match self {
   97 +            Self::Index(idx) => {
   98 +                let idx = Self::convert_index(idx, length);
   99 +                if idx >= 0 && idx < length {
  100 +                    indices.insert(idx as usize);
  101 +                }
  102 +            }
  103 +            Self::Slice((start, end)) => {
  104 +                let start_idx = Self::convert_index(start, length);
  105 +                let end_idx = Self::convert_index(end, length);
  106 +
  107 +                let start_idx = if start_idx < 0 { 0 } else { start_idx as usize };
  108 +                let end_idx = if end_idx >= length {
  109 +                    (length - 1) as usize
  110 +                } else {
  111 +                    end_idx as usize
  112 +                };
  113 +                for idx in start_idx..=end_idx {
  114 +                    indices.insert(idx);
  115 +                }
  116 +            }
  117 +        }
  118 +        indices
  119 +    }
  120 +
  121 +    fn convert_index(index: &Index, length: i32) -> i32 {
  122 +        match index {
  123 +            Index::Index(idx) => *idx,
  124 +            Index::LastIndex(idx) => length + *idx - 1,
  125 +        }
  126 +    }
  127 +}

  129 +/// Represents the end level in hierarchical structure.
  130 +#[derive(Debug, Clone, PartialEq)]
  131 +pub enum RecursiveLevelEnd {
  132 +    /// Specifies the end of the recursive level.
  133 +    Index(u8),
  134 +    /// Specifies that the recursion should continue to the last level.
  135 +    Last,
  136 +}
  137 +
  138 +/// Represents the selected levels in hierarchical structure.
  139 +#[derive(Debug, Clone, PartialEq)]
  140 +pub struct RecursiveLevel {
  141 +    /// The starting level of the recursive level.
  142 +    pub start: u8,
  143 +    /// The optional end of the recursive level. If None, the level applies only to the start level.
  144 +    pub end: Option<RecursiveLevelEnd>,
  145 +}
  146 +
  147 +impl RecursiveLevel {
  148 +    /// Checks if the current level matches the recursive level.
  149 +    ///
  150 +    /// # Arguments
  151 +    ///
  152 +    /// * `level` - The current level in hierarchical structure.
  153 +    ///
  154 +    /// # Returns
  155 +    ///
  156 +    /// A tuple (is_match, should_continue):
  157 +    /// - is_match: Indicates whether the current level matches the level criteria.
  158 +    /// - should_continue: Indicates whether to continue processing data at the next level.
  159 +    pub(crate) fn check_recursive_level(&self, level: u8) -> (bool, bool) {
  160 +        if let Some(end) = &self.end {
  161 +            match end {
  162 +                RecursiveLevelEnd::Index(end) => {
  163 +                    if level < self.start && self.start <= *end {
  164 +                        (false, true)
  165 +                    } else if level >= self.start && level <= *end {
  166 +                        (true, true)
  167 +                    } else {
  168 +                        (false, false)
  169 +                    }
  170 +                }
  171 +                RecursiveLevelEnd::Last => {
  172 +                    if level < self.start {
  173 +                        (false, true)
  174 +                    } else {
  175 +                        (true, true)
  176 +                    }
  177 +                }
  178 +            }
  179 +        } else if level < self.start {
  180 +            (false, true)
  181 +        } else if level == self.start {
  182 +            (true, false)
  183 +        } else {
  184 +            (false, false)
  185 +        }
  186 +    }
  187 +}

  189 +/// Represents a literal value used in filter expression.
  190 +#[derive(Debug, Clone)]
  191 +pub enum PathValue<'a> {
  192 +    /// Null value.
  193 +    Null,
  194 +    /// Boolean value.
  195 +    Boolean(bool),
  196 +    /// Number value.
  197 +    Number(Number),
  198 +    /// UTF-8 string.
  199 +    String(Cow<'a, str>),
  200 +    /// RawJsonb (Array or Object) value, can't be used for calculation.
  201 +    Raw(RawJsonb<'a>),
  202 +}
  203 +
  204 +impl PartialOrd for PathValue<'_> {
  205 +    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
  206 +        match (self, other) {
  207 +            (PathValue::Null, PathValue::Null) => Some(Ordering::Equal),
  208 +            (PathValue::Boolean(l), PathValue::Boolean(r)) => l.partial_cmp(r),
  209 +            (PathValue::Number(l), PathValue::Number(r)) => l.partial_cmp(r),
  210 +            (PathValue::String(l), PathValue::String(r)) => l.partial_cmp(r),
  211 +            (_, _) => None,
  212 +        }
  213 +    }
  214 +}
  215 +
  216 +impl PartialEq for PathValue<'_> {
  217 +    fn eq(&self, other: &Self) -> bool {
  218 +        self.partial_cmp(other) == Some(Ordering::Equal)
  219 +    }
  220 +}

  263 +/// Represents a filter expression used to filter Array or Object.
  264 +#[derive(Debug, Clone, PartialEq)]
  265 +pub enum Expr<'a> {
  266 +    /// JSON Path chains.
  267 +    Paths(Vec<Path<'a>>),
  268 +    /// Literal value.
  269 +    Value(Box<PathValue<'a>>),
  270 +    /// Filter expression that performs a binary operation, returns a boolean value,
  271 +    /// Or arithmetic expression that performs an arithmetic operation, returns a number value.
  272 +    BinaryOp {
  273 +        op: BinaryOperator,
  274 +        left: Box<Self>,
  275 +        right: Box<Self>,

  278 +    UnaryOp {
  279 +        op: UnaryOperator,

  281 +    },
  282 +    /// Filter function, returns a boolean value.
  283 +    ExistsFunc(Vec<Path<'a>>),

  350 +impl Display for Path<'_> {
  351 +    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
  352 +        match self {
  353 +            Path::Root => {
  354 +                write!(f, "$")?;
  355 +            }
  356 +            Path::Current => {
  357 +                write!(f, "@")?;
  358 +            }
  359 +            Path::DotWildcard => {
  360 +                write!(f, ".*")?;
  361 +            }
  362 +            Path::RecursiveDotWildcard(level_opt) => {
  363 +                write!(f, ".**")?;
  364 +                if let Some(level) = level_opt {
  365 +                    write!(f, "{{")?;
  366 +                    write!(f, "{level}")?;
  367 +                    write!(f, "}}")?;
  368 +                }
  369 +            }
  370 +            Path::BracketWildcard => {
  371 +                write!(f, "[*]")?;
  372 +            }
  373 +            Path::ColonField(field) => {
  374 +                write!(f, ":{field}")?;
  375 +            }
  376 +            Path::DotField(field) => {
  377 +                write!(f, ".{field}")?;
  378 +            }
  379 +            Path::ObjectField(field) => {
  380 +                write!(f, "[\"{field}\"]")?;
  381 +            }
  382 +            Path::ArrayIndices(indices) => {
  383 +                write!(f, "[")?;
  384 +                for (i, index) in indices.iter().enumerate() {
  385 +                    if i > 0 {
  386 +                        write!(f, ", ")?;
  387 +                    }
  388 +                    write!(f, "{index}")?;
  389 +                }
  390 +                write!(f, "]")?;
  391 +            }
  392 +            Path::FilterExpr(expr) => {
  393 +                write!(f, "?({expr})")?;
  394 +            }
  395 +            Path::Expr(expr) => {
  396 +                write!(f, "{expr}")?;
  397 +            }
  398 +        }
  399 +        Ok(())
  400 +    }
  401 +}

  491 +impl Display for Expr<'_> {
  492 +    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
  493 +        match self {
  494 +            Expr::Paths(paths) => {
  495 +                for path in paths {
  496 +                    write!(f, "{path}")?;
  497 +                }
  498 +            }
  499 +            Expr::Value(v) => {
  500 +                write!(f, "{v}")?;
  501 +            }
  502 +            Expr::BinaryOp { op, left, right } => {
  503 +                if let Expr::BinaryOp { op: left_op, .. } = &**left {
  504 +                    if left_op == &BinaryOperator::And || left_op == &BinaryOperator::Or {
  505 +                        write!(f, "({left})")?;
  506 +                    } else {
  507 +                        write!(f, "{left}")?;
  508 +                    }
  509 +                } else {
  510 +                    write!(f, "{left}")?;
  511 +                }
  512 +                write!(f, " {op} ")?;
  513 +                if let Expr::BinaryOp { op: right_op, .. } = &**right {
  514 +                    if right_op == &BinaryOperator::And || right_op == &BinaryOperator::Or {
  515 +                        write!(f, "({right})")?;
  516 +                    } else {
  517 +                        write!(f, "{right}")?;
  518 +                    }
  519 +                } else {
  520 +                    write!(f, "{right}")?;
  521 +                }
  522 +            }
  523 +            Expr::UnaryOp { op, operand } => {
  524 +                write!(f, "{}{}", op, operand)?;
  525 +            }
  526 +            Expr::ExistsFunc(paths) => {
  527 +                f.write_str("exists(")?;
  528 +                for path in paths {
  529 +                    write!(f, "{path}")?;
  530 +                }
  531 +                f.write_str(")")?;
  532 +            }
  533 +        }
  534 +        Ok(())
  535 +    }
