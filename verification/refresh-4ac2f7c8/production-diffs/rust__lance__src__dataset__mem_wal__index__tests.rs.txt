   77 +    fn can_answer(&self, query: &dyn MemQuery) -> bool {
   78 +        matches!(
   79 +            query.as_any().downcast_ref::<SargableQuery>(),
   80 +            Some(SargableQuery::Equals(_) | SargableQuery::IsIn(_))
   81 +        )
   82 +    }

   84 +    fn insert(&self, batch: &RecordBatch, row_offset: RowPosition) -> Result<()> {
   85 +        let array = &batch[self.columns[0].as_str()];
   86 +        let mut postings = self.postings.write().unwrap();
   87 +        for row in 0..array.len() {
   88 +            if let Some(key) = Self::key(array.as_ref(), row) {
   89 +                postings
   90 +                    .entry(key)
   91 +                    .or_default()
   92 +                    .push(row_offset + row as u64);
   93 +            }
   94 +        }
   95 +        Ok(())
   96 +    }
   97 +
   98 +    fn resident_bytes(&self) -> usize {
   99 +        let postings = self.postings.read().unwrap();
  100 +        postings
  101 +            .iter()
  102 +            .map(|(key, positions)| {
  103 +                key.len() + positions.len() * std::mem::size_of::<RowPosition>()
  104 +            })
  105 +            .sum()
  106 +    }
  107 +
  108 +    fn search(&self, query: &dyn MemQuery, ctx: &SearchContext) -> Result<Option<MemMatches>> {
  109 +        let Some(query) = query.as_any().downcast_ref::<SargableQuery>() else {
  110 +            return Ok(None);
  111 +        };
  112 +        let postings = self.postings.read().unwrap();
  113 +        let hits: Vec<RowPosition> = match query {
  114 +            SargableQuery::Equals(value) => postings
  115 +                .get(&value.to_string())
  116 +                .cloned()
  117 +                .unwrap_or_default(),
  118 +            SargableQuery::IsIn(values) => values
  119 +                .iter()
  120 +                .filter_map(|value| postings.get(&value.to_string()))
  121 +                .flatten()
  122 +                .copied()
  123 +                .collect(),
  124 +            _ => return Ok(None),
  125 +        };
  126 +        Ok(Some(MemMatches::exact(
  127 +            hits.into_iter().filter(|p| *p <= ctx.max_visible),
  128 +        )))
  129 +    }
