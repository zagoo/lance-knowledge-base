 2215 +fn infer_generic_run_count(
 2216 +    source: BlockRleRunCount,
 2217 +    values_payload: Option<&LanceBuffer>,
 2218 +    lengths_payload: Option<&LanceBuffer>,
 2219 +    bits_per_value: u64,
 2220 +    num_values: u64,
 2221 +) -> Result<u64> {
 2222 +    let run_count = match source {
 2223 +        BlockRleRunCount::ConstantRunLength(length) => {
 2224 +            if length == 0 || !num_values.is_multiple_of(length) {
 2225 +                return Err(Error::invalid_input(format!(
 2226 +                    "RLE constant run length {length} does not divide {num_values} values"
 2227 +                )));
 2228 +            }
 2229 +            num_values / length

 2231 +        BlockRleRunCount::RangeRunLengths { start, step } => {
 2232 +            if start == 0 || step == 0 {
 2233 +                return Err(Error::invalid_input(
 2234 +                    "RLE range run lengths must be positive",
 2235 +                ));
 2236 +            }

 2238 +            let mut low = 1_u64;
 2239 +            let mut high = num_values;
 2240 +            let mut found = None;
 2241 +            while low <= high {
 2242 +                let count = low + (high - low) / 2;
 2243 +                let count128 = u128::from(count);
 2244 +                let sum = count128
 2245 +                    .checked_mul(

 2247 +                            .checked_mul(2)
 2248 +                            .and_then(|start| {
 2249 +                                u128::from(count - 1)
 2250 +                                    .checked_mul(u128::from(step))
 2251 +                                    .and_then(|tail| start.checked_add(tail))
 2252 +                            })
 2253 +                            .unwrap_or(u128::MAX),
 2254 +                    )
 2255 +                    .map(|sum| sum / 2)
 2256 +                    .unwrap_or(u128::MAX);

 2258 +                    std::cmp::Ordering::Less => low = count + 1,
 2259 +                    std::cmp::Ordering::Greater => high = count - 1,
 2260 +                    std::cmp::Ordering::Equal => {
 2261 +                        found = Some(count);
 2262 +                        break;
 2263 +                    }
 2264 +                }
 2265 +            }
 2266 +            found.ok_or_else(|| {

 2285 +            num_values,
