  117 +/// Represents a decimal number with 64-bit precision.
  118 +///
  119 +/// This structure stores a decimal value as an integer with a scale factor,
  120 +/// allowing for precise representation of decimal numbers without floating-point errors.
  121 +/// The scale indicates how many decimal places the value has (e.g., scale=2 means 2 decimal places).
  122 +#[derive(Debug, Clone)]
  123 +pub struct Decimal64 {
  124 +    /// Number of decimal places (e.g., 2 for values like 123.45)
  125 +    pub scale: u8,
  126 +    /// The actual value, scaled by 10^scale (e.g., 12345 for 123.45 with scale=2)
  127 +    pub value: i64,
  128 +}
  129 +
  130 +impl Decimal64 {
  131 +    /// Converts the decimal value to a floating-point representation.
  132 +    ///
  133 +    /// This is useful when you need to perform floating-point operations,
  134 +    /// but note that it may introduce precision loss inherent to floating-point arithmetic.
  135 +    pub(crate) fn to_float64(&self) -> f64 {
  136 +        let div = 10_f64.powi(self.scale as i32);
  137 +        self.value as f64 / div
  138 +    }
  139 +}

  271 +    {
  272 +        match self {
  273 +            Self::Int64(v) => serializer.serialize_i64(*v),
  274 +            Self::UInt64(v) => serializer.serialize_u64(*v),
  275 +            Self::Float64(v) => serializer.serialize_f64(*v),
  276 +            Self::Decimal64(_) | Self::Decimal128(_) | Self::Decimal256(_) => {
  277 +                let (value, scale) = match self {
  278 +                    Self::Decimal64(v) => (v.value as f64, v.scale as i32),
  279 +                    Self::Decimal128(v) => (v.value as f64, v.scale as i32),
  280 +                    Self::Decimal256(v) => (v.value.as_f64(), v.scale as i32),
  281 +                    _ => unreachable!(),
  282 +                };
  283 +                let scaled_value = value / 10f64.powi(scale);
  284 +                serializer.serialize_f64(scaled_value)
  285 +            }
  286 +        }
  287 +    }

  571 +                        .checked_rem(b)
  572 +                        .map(Number::UInt64)
  573 +                        .ok_or(Error::Message("UInt64 overflow".to_string()))
  574 +                }
  575 +            }
  576 +            (Self::UInt64(a), Self::Int64(b)) => {
  577 +                if b == 0 {
  578 +                    return Err(Error::Message("Division by zero".to_string()));
  579 +                }
  580 +                if b < 0 {
  581 +                    (*a as i64)
  582 +                        .checked_rem(b)
  583 +                        .map(Number::Int64)
  584 +                        .ok_or(Error::Message("Int64 overflow".to_string()))
  585 +                } else {
  586 +                    a.checked_rem(b as u64)
  587 +                        .map(Number::UInt64)
  588 +                        .ok_or(Error::Message("UInt64 overflow".to_string()))
  589 +                }
  590 +            }
  591 +            (Self::Float64(a), Self::Float64(b)) => {
  592 +                if b == 0.0 {
  593 +                    return Err(Error::Message("Division by zero".to_string()));
  594 +                }
  595 +                Ok(Self::Float64(a % b))
  596 +            }
  597 +            (a, b) => {
  598 +                let a_float = a.as_f64();
  599 +                let b_float = b.as_f64();
  600 +                if b_float == 0.0 {
  601 +                    return Err(Error::Message("Division by zero".to_string()));
  602 +                }
  603 +                Ok(Self::Float64(a_float % b_float))
  604 +            }
  605 +        }
  606 +    }
  607 +}

  609 +fn exact_float_to_int<T>(value: f64) -> Option<T>
  610 +where
  611 +    T: Bounded + AsPrimitive<f64>,
  612 +    f64: AsPrimitive<T>,
  613 +{
  614 +    if !value.is_finite() || value.fract() != 0.0 {
  615 +        return None;
  616 +    }
  617 +
  618 +    if fits_float_target_range::<T>(value) {
  619 +        Some(value.as_())
  620 +    } else {
  621 +        None
  622 +    }
  623 +}

  625 +fn round_float_to_int<T>(value: f64) -> Option<T>
  626 +where
  627 +    T: Bounded + AsPrimitive<f64>,
  628 +    f64: AsPrimitive<T>,
  629 +{
  630 +    if !value.is_finite() {
  631 +        return None;
  632 +    }
  633 +
  634 +    let rounded = value.round();
  635 +    if fits_float_target_range::<T>(rounded) {
  636 +        Some(rounded.as_())
  637 +    } else {
  638 +        None
  639 +    }
  640 +}

  642 +// Use a half-open range so the positive bound can be expressed as `-min` for
  643 +// signed integers, which avoids relying on an imprecise `MAX as f64`.
  644 +#[inline]
  645 +fn fits_float_target_range<T>(value: f64) -> bool
  646 +where
  647 +    T: Bounded + AsPrimitive<f64>,
  648 +{
  649 +    let lower = T::min_value().as_();
  650 +    let upper_exclusive = if lower < 0.0 {
  651 +        -lower
  652 +    } else {
  653 +        T::max_value().as_() + 1.0
  654 +    };
  655 +
  656 +    value >= lower && value < upper_exclusive
  657 +}

  659 +fn exact_decimal_i128(value: i128, scale: u8) -> Option<i128> {
  660 +    if scale == 0 {
  661 +        return Some(value);
  662 +    }
  663 +
  664 +    let Some(divisor) = I128_POWERS_OF_10.get(scale as usize) else {
  665 +        return if value == 0 { Some(0) } else { None };
  666 +    };
  667 +
  668 +    if value % divisor == 0 {
  669 +        Some(value / divisor)
  670 +    } else {
  671 +        None
  672 +    }
  673 +}

  675 +fn round_decimal_i128(value: i128, scale: u8) -> Option<i128> {
  676 +    if scale == 0 {
  677 +        return Some(value);
  678 +    }
  679 +
  680 +    let Some(divisor) = I128_POWERS_OF_10.get(scale as usize) else {
  681 +        // 10^scale no longer fits in i128, so any i128-backed decimal rounds to 0.
  682 +        return Some(0);
  683 +    };
  684 +
  685 +    let quotient = value / divisor;
  686 +    let remainder = value % divisor;
  687 +    let abs_remainder = remainder.unsigned_abs();
  688 +    let divisor = *divisor as u128;
  689 +    let round_up = abs_remainder >= divisor - abs_remainder;
  690 +
  691 +    if !round_up {
  692 +        Some(quotient)
  693 +    } else if value >= 0 {
  694 +        quotient.checked_add(1)
  695 +    } else {
  696 +        quotient.checked_sub(1)
  697 +    }
  698 +}

  702 +    let value = i256::try_from(abs).ok()?;
  703 +    if negative {
  704 +        i256::ZERO.checked_sub(value)

  710 +fn exact_decimal_i256(value: i256, scale: u8) -> Option<i256> {
  711 +    if scale == 0 {
  712 +        return Some(value);
  713 +    }
  714 +
  715 +    let Some(divisor) = U256::new(10).checked_pow(scale as u32) else {
  716 +        return if value == i256::ZERO {
  717 +            Some(i256::ZERO)
  718 +        } else {
  719 +            None
  720 +        };
  721 +    };
  722 +
  723 +    let negative = value < i256::ZERO;
  724 +    let abs_value = value.unsigned_abs();
  725 +    let quotient = abs_value / divisor;
  726 +    let remainder = abs_value % divisor;
  727 +    if remainder != U256::ZERO {
  728 +        return None;
  729 +    }
  730 +
  731 +    signed_i256_from_abs(quotient, negative)
  732 +}

  734 +fn round_decimal_i256(value: i256, scale: u8) -> Option<i256> {
  735 +    if scale == 0 {
  736 +        return Some(value);
  737 +    }
  738 +
  739 +    let Some(divisor) = U256::new(10).checked_pow(scale as u32) else {
  740 +        // 10^scale exceeds the i256 range by enough that the rounded result is always 0.
  741 +        return Some(i256::ZERO);
  742 +    };
  743 +
  744 +    let negative = value < i256::ZERO;
  745 +    let abs_value = value.unsigned_abs();
  746 +    let quotient = abs_value / divisor;
  747 +    let remainder = abs_value % divisor;
  748 +    let round_up = remainder >= divisor - remainder;
  749 +    let rounded = if round_up {
  750 +        quotient.checked_add(U256::new(1))?
  751 +    } else {
  752 +        quotient
  753 +    };
  754 +
  755 +    signed_i256_from_abs(rounded, negative)
  756 +}

  830 +impl Ord for Number {
  831 +    /// Returns the ordering of the number compared to another number.
  832 +    ///
  833 +    /// This method implements precise comparison between different number types:
  834 +    /// - When comparing decimal types with other types, it converts the non-decimal type
  835 +    ///   to a decimal representation to preserve precision
  836 +    /// - When comparing between decimal types of different precision, it upgrades the lower
  837 +    ///   precision decimal to match the higher precision one
  838 +    /// - Only falls back to floating-point comparison as a last resort
  839 +    #[inline]
  840 +    fn cmp(&self, other: &Self) -> Ordering {
  841 +        match (self, other) {
  842 +            // Same type comparisons - use native comparison
  843 +            (Self::Int64(l), Self::Int64(r)) => l.cmp(r),
  844 +            (Self::UInt64(l), Self::UInt64(r)) => l.cmp(r),
  845 +            (Self::Float64(l), Self::Float64(r)) => OrderedFloat(*l).cmp(&OrderedFloat(*r)),
  846 +            (Self::Decimal64(l), Self::Decimal64(r)) => {
  847 +                // Compare decimal values with the same scale
  848 +                if l.scale == r.scale {
  849 +                    l.value.cmp(&r.value)
  850 +                } else {
  851 +                    // Adjust scales to match for proper comparison
  852 +                    if let Some((l_val, r_val)) =
  853 +                        adjust_decimal_scales(l.value as i128, l.scale, r.value as i128, r.scale)
  854 +                    {
  855 +                        l_val.cmp(&r_val)
  856 +                    } else {
  857 +                        let l = OrderedFloat(self.as_f64());
  858 +                        let r = OrderedFloat(other.as_f64());
  859 +                        l.cmp(&r)
  860 +                    }
  861 +                }
  862 +            }
  863 +            (Self::Decimal128(l), Self::Decimal128(r)) => {
  864 +                // Compare decimal values with the same scale
  865 +                if l.scale == r.scale {
  866 +                    l.value.cmp(&r.value)
  867 +                } else {
  868 +                    // Adjust scales to match for proper comparison
  869 +                    if let Some((l_val, r_val)) =
  870 +                        adjust_decimal_scales(l.value, l.scale, r.value, r.scale)
  871 +                    {
  872 +                        l_val.cmp(&r_val)
  873 +                    } else {
  874 +                        let l = OrderedFloat(self.as_f64());
  875 +                        let r = OrderedFloat(other.as_f64());
  876 +                        l.cmp(&r)
  877 +                    }
  878 +                }
  879 +            }
  880 +            (Self::Decimal256(l), Self::Decimal256(r)) => {
  881 +                // Compare decimal values with the same scale
  882 +                if l.scale == r.scale {
  883 +                    l.value.cmp(&r.value)
  884 +                } else {
  885 +                    // For i256 comparison with different scales, we need to adjust manually
  886 +                    let scale_diff = l.scale as i32 - r.scale as i32;
  887 +                    if scale_diff > 0 {
  888 +                        // l has more decimal places, scale up r
  889 +                        let scale_factor = i256::from(10).pow(scale_diff as u32);
  890 +                        if let Some(r_val) = r.value.checked_mul(scale_factor) {
  891 +                            return l.value.cmp(&r_val);
  892 +                        }
  893 +                    } else {
  894 +                        // r has more decimal places, scale up l
  895 +                        let scale_factor = i256::from(10).pow((-scale_diff) as u32);
  896 +                        if let Some(l_val) = l.value.checked_mul(scale_factor) {
  897 +                            return l_val.cmp(&r.value);
  898 +                        }
  899 +                    }
  900 +                    // multiply overflow, fallback to used float compare
  901 +                    let l = OrderedFloat(self.as_f64());
  902 +                    let r = OrderedFloat(other.as_f64());
  903 +                    l.cmp(&r)
  904 +                }
  905 +            }
  906 +
  907 +            // Integer to integer comparisons
  908 +            (Self::Int64(l), Self::UInt64(r)) => {
  909 +                if *l < 0 {
  910 +                    Ordering::Less
  911 +                } else {
  912 +                    (*l as u64).cmp(r)
  913 +                }
  914 +            }
  915 +            (Self::UInt64(l), Self::Int64(r)) => {
  916 +                if *r < 0 {
  917 +                    Ordering::Greater
  918 +                } else {
  919 +                    l.cmp(&(*r as u64))
  920 +                }
  921 +            }
  922 +
  923 +            // Decimal64 comparisons with other types
  924 +            (Self::Decimal64(_), Self::Int64(r)) => {

  926 +                let r_decimal = Decimal64 {
  927 +                    scale: 0,
  928 +                    value: *r,
  929 +                };
  930 +                self.cmp(&Self::Decimal64(r_decimal))

  932 +            (Self::Int64(l), Self::Decimal64(_)) => {

 1105 +/// Helper function to adjust decimal scales for comparison
 1106 +///
 1107 +/// Given two decimal values with potentially different scales,
 1108 +/// this function adjusts them to have the same scale for accurate comparison.
 1109 +fn adjust_decimal_scales(
 1110 +    l_val: i128,
 1111 +    l_scale: u8,
 1112 +    r_val: i128,
 1113 +    r_scale: u8,
 1114 +) -> Option<(i128, i128)> {
 1115 +    let scale_diff = l_scale as i32 - r_scale as i32;
 1116 +
 1117 +    match scale_diff.cmp(&0) {
 1118 +        Ordering::Greater => {
 1119 +            // l has more decimal places, scale up r
 1120 +            let scale_factor = 10_i128.pow(scale_diff as u32);
 1121 +            let r_val = r_val.checked_mul(scale_factor)?;
 1122 +            Some((l_val, r_val))
 1123 +        }
 1124 +        Ordering::Less => {
 1125 +            // r has more decimal places, scale up l
 1126 +            let scale_factor = 10_i128.pow((-scale_diff) as u32);
 1127 +            let l_val = l_val.checked_mul(scale_factor)?;
 1128 +            Some((l_val, r_val))
 1129 +        }
 1130 +        Ordering::Equal => {
 1131 +            // Same scale, no adjustment needed
 1132 +            Some((l_val, r_val))
 1133 +        }
 1134 +    }
 1135 +}
