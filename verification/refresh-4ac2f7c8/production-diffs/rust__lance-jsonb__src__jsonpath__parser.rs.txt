   41 +/// Parsing the input string to JSON Path.
   42 +pub fn parse_json_path(input: &[u8]) -> Result<JsonPath<'_>, Error> {
   43 +    match json_path(input) {
   44 +        Ok((rest, json_path)) => {
   45 +            if !rest.is_empty() {
   46 +                return Err(Error::InvalidJsonPath);
   47 +            }
   48 +            Ok(json_path)
   49 +        }
   50 +        Err(nom::Err::Error(_) | nom::Err::Failure(_)) => Err(Error::InvalidJsonPath),
   51 +        Err(nom::Err::Incomplete(_)) => unreachable!(),
   52 +    }
   53 +}

   63 +fn check_escaped(input: &[u8], i: &mut usize) -> bool {
   64 +    if *i + 1 >= input.len() {
   65 +        return false;
   66 +    }
   67 +    if input[*i + 1] == b'u' {
   68 +        if *i + 5 >= input.len() {
   69 +            return false;
   70 +        }
   71 +        if input[*i + 2] == b'{' {
   72 +            if *i + 7 >= input.len() {
   73 +                return false;
   74 +            }
   75 +            *i += UNICODE_LEN + 4;
   76 +        } else {
   77 +            *i += UNICODE_LEN + 2;
   78 +        }
   79 +    } else {
   80 +        *i += 2;
   81 +    }
   82 +    true
   83 +}

   85 +pub fn raw_string(input: &[u8]) -> IResult<&[u8], Cow<'_, str>> {
   86 +    let mut i = 0;
   87 +    let mut escapes = 0;
   88 +    while i < input.len() {
   89 +        let c = input[i];
   90 +        match c {
   91 +            b'\\' => {
   92 +                escapes += 1;
   93 +                if !check_escaped(input, &mut i) {
   94 +                    return Err(nom::Err::Error(NomError::new(input, ErrorKind::Char)));
   95 +                }
   96 +            }
   97 +            b' ' | b',' | b'.' | b':' | b'{' | b'}' | b'[' | b']' | b'(' | b')' | b'?' | b'@'
   98 +            | b'$' | b'|' | b'<' | b'>' | b'!' | b'=' | b'+' | b'-' | b'*' | b'/' | b'%' | b'"'
   99 +            | b'\'' => {
  100 +                break;
  101 +            }
  102 +            _ => {
  103 +                i += 1;
  104 +            }
  105 +        }
  106 +    }
  107 +    if i > 0 {
  108 +        if escapes == 0 {
  109 +            if let Ok(s) = std::str::from_utf8(&input[..i]) {
  110 +                return Ok((&input[i..], Cow::Borrowed(s)));
  111 +            }
  112 +        } else {
  113 +            let data = &input[..i];
  114 +            let len = i - escapes;
  115 +            let mut idx = 0;
  116 +            let s = parse_string(data, len, &mut idx)
  117 +                .map_err(|_| nom::Err::Error(NomError::new(input, ErrorKind::Char)))?;
  118 +            return Ok((&input[i..], Cow::Owned(s)));
  119 +        }
  120 +    }
  121 +    Err(nom::Err::Error(NomError::new(input, ErrorKind::Char)))
  122 +}

  124 +pub fn string(input: &[u8]) -> IResult<&[u8], Cow<'_, str>> {
  125 +    if input.is_empty() || input[0] != b'"' {
  126 +        return Err(nom::Err::Error(NomError::new(input, ErrorKind::Char)));
  127 +    }
  128 +    let mut i = 1;
  129 +    let mut escapes = 0;
  130 +    while i < input.len() {
  131 +        let c = input[i];
  132 +        match c {
  133 +            b'\\' => {
  134 +                escapes += 1;
  135 +                if !check_escaped(input, &mut i) {
  136 +                    return Err(nom::Err::Error(NomError::new(input, ErrorKind::Char)));
  137 +                }
  138 +            }
  139 +            b'"' => {
  140 +                break;
  141 +            }
  142 +            _ => {
  143 +                i += 1;
  144 +            }
  145 +        }
  146 +    }
  147 +    if i > 1 {
  148 +        if escapes == 0 {
  149 +            if let Ok(s) = std::str::from_utf8(&input[1..i]) {
  150 +                return Ok((&input[i + 1..], Cow::Borrowed(s)));
  151 +            }
  152 +        } else {
  153 +            let data = &input[1..i];
  154 +            let len = i - 1 - escapes;
  155 +            let mut idx = 1;
  156 +            let s = parse_string(data, len, &mut idx)
  157 +                .map_err(|_| nom::Err::Error(NomError::new(input, ErrorKind::Char)))?;
  158 +            return Ok((&input[i + 1..], Cow::Owned(s)));
  159 +        }
  160 +    }
  161 +    Err(nom::Err::Error(NomError::new(input, ErrorKind::Char)))
  162 +}

  180 +fn recursive_level(input: &[u8]) -> IResult<&[u8], RecursiveLevel> {

  195 +                end: None,
  196 +            },
  197 +        ),

  339 +fn paths(input: &[u8]) -> IResult<&[u8], Vec<Path<'_>>> {
  340 +    map(
  341 +        pair(opt(pre_path), many0(path)),
  342 +        |(opt_pre_path, mut paths)| {
  343 +            if let Some(pre_path) = opt_pre_path {
  344 +                paths.insert(0, pre_path);
  345 +            }
  346 +            paths
  347 +        },
  348 +    )

  352 +fn expr_paths(input: &[u8], is_root_expr: bool) -> IResult<&[u8], Vec<Path<'_>>> {
  353 +    let parse_current = map_res(
  354 +        cond(!is_root_expr, value(Path::Current, char('@'))),
  355 +        |res| match res {
  356 +            Some(v) => Ok(v),
  357 +            None => Err(NomError::new(input, ErrorKind::Char)),
  358 +        },
  359 +    );
  360 +    map(
  361 +        pair(
  362 +            alt((value(Path::Root, char('$')), parse_current)),
  363 +            many0(delimited(multispace0, inner_path, multispace0)),
  364 +        ),
  365 +        |(pre_path, mut paths)| {
  366 +            paths.insert(0, pre_path);
  367 +            paths
  368 +        },
  369 +    )
  370 +    .parse(input)
  371 +}

  433 +fn expr_atom(input: &[u8], is_root_expr: bool) -> IResult<&[u8], Expr<'_>> {
  434 +    alt((
  435 +        map(
  436 +            pair(
  437 +                unary_arith_op,
  438 +                delimited(multispace0, |i| inner_expr(i, is_root_expr), multispace0),
  439 +            ),
  440 +            |(op, operand)| Expr::UnaryOp {
  441 +                op,
  442 +                operand: Box::new(operand),
  443 +            },
  444 +        ),
  445 +        map(
  446 +            pair(
  447 +                pair(
  448 +                    delimited(multispace0, |i| inner_expr(i, is_root_expr), multispace0),
  449 +                    op,
  450 +                ),
  451 +                delimited(multispace0, |i| inner_expr(i, is_root_expr), multispace0),
  452 +            ),
  453 +            |((left, op), right)| Expr::BinaryOp {
  454 +                op,
  455 +                left: Box::new(left),
  456 +                right: Box::new(right),
  457 +            },
  458 +        ),
  459 +        map(
  460 +            delimited(
  461 +                terminated(char('('), multispace0),
  462 +                |i| expr_or(i, is_root_expr),
  463 +                preceded(multispace0, char(')')),
  464 +            ),
  465 +            |expr| expr,
  466 +        ),
  467 +        map(exists_func, Expr::ExistsFunc),
  468 +    ))
  469 +    .parse(input)

  504 +fn expr_and(input: &[u8], is_root_expr: bool) -> IResult<&[u8], Expr<'_>> {
  505 +    map(
  506 +        separated_list1(delimited(multispace0, tag("&&"), multispace0), |i| {
  507 +            expr_atom(i, is_root_expr)
  508 +        }),
  509 +        |exprs| {
  510 +            let mut expr = exprs[0].clone();
  511 +            for right in exprs.iter().skip(1) {
  512 +                expr = Expr::BinaryOp {
  513 +                    op: BinaryOperator::And,
  514 +                    left: Box::new(expr),
  515 +                    right: Box::new(right.clone()),
  516 +                };
  517 +            }
  518 +            expr
  519 +        },
  520 +    )
  521 +    .parse(input)
  522 +}
  523 +
  524 +fn expr_or(input: &[u8], is_root_expr: bool) -> IResult<&[u8], Expr<'_>> {
  525 +    map(
  526 +        separated_list1(delimited(multispace0, tag("||"), multispace0), |i| {
  527 +            expr_and(i, is_root_expr)
  528 +        }),
  529 +        |exprs| {
  530 +            let mut expr = exprs[0].clone();
  531 +            for right in exprs.iter().skip(1) {
  532 +                expr = Expr::BinaryOp {
  533 +                    op: BinaryOperator::Or,
  534 +                    left: Box::new(expr),
  535 +                    right: Box::new(right.clone()),
  536 +                };
  537 +            }
  538 +            expr
  539 +        },
  540 +    )
  541 +    .parse(input)
