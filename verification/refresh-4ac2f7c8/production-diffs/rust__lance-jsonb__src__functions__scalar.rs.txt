  142 +    /// * `Err(Error::InvalidCast)` - If the value cannot be converted to a boolean.
  143 +    /// * `Err(Error)` - If the JSONB data is invalid.
  144 +    ///
  145 +    /// # Examples
  146 +    ///
  147 +    /// ```rust
  148 +    /// use lance_jsonb::OwnedJsonb;
  149 +    ///
  150 +    /// // Boolean values

  153 +    ///

  155 +    /// assert!(!false_jsonb.as_raw().to_bool().unwrap());

  158 +    /// let true_str = r#""true""#.parse::<OwnedJsonb>().unwrap();
  159 +    /// assert!(true_str.as_raw().to_bool().unwrap());
  160 +    ///

  162 +    /// assert!(!false_str.as_raw().to_bool().unwrap());
  163 +    ///
  164 +    /// let true_str_lowercase = r#""TRUE""#.parse::<OwnedJsonb>().unwrap();
  165 +    /// assert!(true_str_lowercase.as_raw().to_bool().unwrap());
  166 +    ///
  167 +    /// // Invalid conversions
  168 +    /// let num_jsonb = "1".parse::<OwnedJsonb>().unwrap();
  169 +    /// let result = num_jsonb.as_raw().to_bool();
  170 +    /// assert!(result.is_err());
  171 +    ///
  172 +    /// let arr_jsonb = "[1, 2, 3]".parse::<OwnedJsonb>().unwrap();
  173 +    /// let result = arr_jsonb.as_raw().to_bool();
  174 +    /// assert!(result.is_err());
  175 +    ///
  176 +    /// let obj_jsonb = r#"{"a": 1}"#.parse::<OwnedJsonb>().unwrap();
  177 +    /// let result = obj_jsonb.as_raw().to_bool();
  178 +    /// assert!(result.is_err());
  179 +    ///
  180 +    /// let null_jsonb = "null".parse::<OwnedJsonb>().unwrap();
  181 +    /// let result = null_jsonb.as_raw().to_bool();
  182 +    /// assert!(result.is_err());
  183 +    ///
  184 +    /// let invalid_str = r#""maybe""#.parse::<OwnedJsonb>().unwrap();
  185 +    /// let result = invalid_str.as_raw().to_bool();
  186 +    /// assert!(result.is_err());
  187 +    ///
  188 +    /// // Invalid JSONB
  189 +    /// let invalid_jsonb = OwnedJsonb::from(vec![1, 2, 3, 4]);
  190 +    /// let invalid_raw_jsonb = invalid_jsonb.as_raw();
  191 +    /// let result = invalid_raw_jsonb.to_bool();
  192 +    /// assert!(result.is_err());
  193 +    /// ```
  194 +    pub fn to_bool(&self) -> Result<bool> {
  195 +        let jsonb_item = JsonbItem::from_raw_jsonb(*self)?;
  196 +        match jsonb_item {
  197 +            JsonbItem::Boolean(v) => {
  198 +                return Ok(v);
  199 +            }
  200 +            JsonbItem::String(s) => {
  201 +                if s.eq_ignore_ascii_case("true") || s.eq_ignore_ascii_case("yes") {
  202 +                    return Ok(true);
  203 +                } else if s.eq_ignore_ascii_case("false") || s.eq_ignore_ascii_case("no") {
  204 +                    return Ok(false);
  205 +                }
  206 +            }
  207 +            _ => {}
  208 +        }
  209 +        Err(Error::InvalidCast)
  210 +    }

  503 +        let jsonb_item = JsonbItem::from_raw_jsonb(*self)?;
  504 +        match jsonb_item {
  505 +            JsonbItem::Boolean(v) => {
  506 +                if v {
  507 +                    return Ok(1_i64);
  508 +                } else {
  509 +                    return Ok(0_i64);
  510 +                }
  511 +            }
  512 +            JsonbItem::Number(num) => {
  513 +                let value = num.as_number()?;
  514 +                if let Some(v) = value.to_i64() {
  515 +                    return Ok(v);
  516 +                }

  519 +                if let Some(v) = parse_string_to_i64(&s) {
  520 +                    return Ok(v);
  521 +                }
  522 +            }
  523 +            _ => {}
  524 +        }
  525 +        Err(Error::InvalidCast)
  526 +    }

  909 +fn parse_string_to_i64(s: &str) -> Option<i64> {
  910 +    if let Ok(v) = s.parse::<i64>() {
  911 +        return Some(v);
  912 +    }
  913 +
  914 +    s.parse::<f64>()
  915 +        .ok()
