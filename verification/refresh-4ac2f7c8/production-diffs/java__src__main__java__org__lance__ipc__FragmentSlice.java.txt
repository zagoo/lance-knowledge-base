   14 +package org.lance.ipc;
   15 +
   16 +import com.google.common.base.MoreObjects;
   17 +import org.apache.arrow.util.Preconditions;
   18 +
   19 +import java.io.Serializable;
   20 +import java.util.Objects;
   21 +
   22 +/** A contiguous range of physical row offsets within a fragment. */
   23 +public final class FragmentSlice implements Serializable {
   24 +  private static final long serialVersionUID = 1L;
   25 +
   26 +  private final int fragmentId;
   27 +  private final long rowOffset;
   28 +  private final long rowCount;
   29 +
   30 +  /**
   31 +   * Creates a fragment slice covering {@code [rowOffset, rowOffset + rowCount)}.
   32 +   *
   33 +   * <p>Row offsets are physical positions in the fragment. Deletions do not compact these offsets,
   34 +   * and deleted rows in the range are omitted by a normal scan. The slice must be used with the
   35 +   * same dataset snapshot from which it was planned.
   36 +   *
   37 +   * @param fragmentId fragment ID in the dataset snapshot
   38 +   * @param rowOffset starting physical row offset in the fragment
   39 +   * @param rowCount number of physical row offsets covered; zero represents an empty slice
   40 +   */
   41 +  public FragmentSlice(int fragmentId, long rowOffset, long rowCount) {
   42 +    Preconditions.checkArgument(
   43 +        fragmentId >= 0, "fragmentId must be non-negative, got %s", fragmentId);
   44 +    Preconditions.checkArgument(
   45 +        rowOffset >= 0, "rowOffset must be non-negative, got %s", rowOffset);
   46 +    Preconditions.checkArgument(rowCount >= 0, "rowCount must be non-negative, got %s", rowCount);
   47 +    Preconditions.checkArgument(
   48 +        rowOffset <= Long.MAX_VALUE - rowCount,
   49 +        "rowOffset + rowCount overflows long: rowOffset=%s, rowCount=%s",
   50 +        rowOffset,
   51 +        rowCount);
   52 +    this.fragmentId = fragmentId;
   53 +    this.rowOffset = rowOffset;
   54 +    this.rowCount = rowCount;
   55 +  }
   56 +
   57 +  public int getFragmentId() {
   58 +    return fragmentId;
   59 +  }
   60 +
   61 +  public long getRowOffset() {
   62 +    return rowOffset;
   63 +  }
   64 +
   65 +  public long getRowCount() {
   66 +    return rowCount;
   67 +  }
   68 +
   69 +  @Override
   70 +  public boolean equals(Object other) {
   71 +    if (this == other) {
   72 +      return true;
   73 +    }
   74 +    if (!(other instanceof FragmentSlice)) {
   75 +      return false;
   76 +    }
   77 +    FragmentSlice that = (FragmentSlice) other;
   78 +    return fragmentId == that.fragmentId
   79 +        && rowOffset == that.rowOffset
   80 +        && rowCount == that.rowCount;
   81 +  }
   82 +
   83 +  @Override
   84 +  public int hashCode() {
   85 +    return Objects.hash(fragmentId, rowOffset, rowCount);
   86 +  }
   87 +
   88 +  @Override
   89 +  public String toString() {
   90 +    return MoreObjects.toStringHelper(this)
   91 +        .add("fragmentId", fragmentId)
   92 +        .add("rowOffset", rowOffset)
   93 +        .add("rowCount", rowCount)
   94 +        .toString();
   95 +  }
   96 +}
