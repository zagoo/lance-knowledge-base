   36 +impl Display for ParseErrorCode {
   37 +    fn fmt(&self, f: &mut core::fmt::Formatter) -> core::fmt::Result {
   38 +        match self {
   39 +            Self::InvalidEOF => f.write_str("EOF while parsing a value"),
   40 +            Self::InvalidNumberValue => f.write_str("invalid number"),
   41 +            Self::InvalidStringValue => f.write_str("invalid string"),
   42 +            Self::ExpectedSomeIdent => f.write_str("expected ident"),
   43 +            Self::ExpectedSomeValue => f.write_str("expected value"),
   44 +            Self::ExpectedColon => f.write_str("expected `:`"),
   45 +            Self::ExpectedArrayCommaOrEnd => f.write_str("expected `,` or `]`"),
   46 +            Self::ExpectedObjectCommaOrEnd => f.write_str("expected `,` or `}`"),
   47 +            Self::UnexpectedTrailingCharacters => f.write_str("trailing characters"),
   48 +            Self::KeyMustBeAString => f.write_str("key must be a string"),
   49 +            Self::ControlCharacterWhileParsingString => {
   50 +                f.write_str("control character (\\u0000-\\u001F) found while parsing a string")
   51 +            }
   52 +            Self::InvalidEscaped(n) => {
   53 +                write!(f, "invalid escaped '{:X}'", n)
   54 +            }
   55 +            Self::InvalidHex(n) => {
   56 +                write!(f, "invalid hex '{:X}'", n)
   57 +            }
   58 +            Self::InvalidLoneLeadingSurrogateInHexEscape(n) => {
   59 +                write!(f, "lone leading surrogate in hex escape '{:X}'", n)
   60 +            }
   61 +            Self::InvalidSurrogateInHexEscape(n) => {
   62 +                write!(f, "invalid surrogate in hex escape '{:X}'", n)
   63 +            }
   64 +            Self::UnexpectedEndOfHexEscape => f.write_str("unexpected end of hex escape"),
   65 +            Self::ObjectDuplicateKey(key) => {
   66 +                write!(f, "duplicate object attribute \"{}\"", key)
   67 +            }
   68 +            Self::ObjectKeyInvalidNumber => f.write_str("object attribute name cannot be a number"),
   69 +            Self::ObjectKeyInvalidCharacter => {
   70 +                f.write_str("object attribute name cannot be invalid character")
   71 +            }
   72 +        }
   73 +    }
   74 +}
