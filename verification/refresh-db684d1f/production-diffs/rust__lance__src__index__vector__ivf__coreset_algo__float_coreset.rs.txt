   26 +    pub(in crate::index::vector::ivf) fn new(

   34 +impl StreamingCoresetAlgorithm for FloatCoresetAlgorithm<'_> {
   35 +    type Coreset = WeightedCoreset;
   36 +
   37 +    fn metric_policy(&self) -> &dyn StreamingKMeansMetricPolicy {
   38 +        self.metric_policy
   39 +    }
   40 +
   41 +    fn prepare_sample(
   42 +        &self,
   43 +        training_data: FixedSizeListArray,
   44 +        sampled_metric: MetricType,
   45 +    ) -> Result<FixedSizeListArray> {
   46 +        self.metric_policy.validate_sample_metric(sampled_metric)?;
   47 +        cast_training_data_to_f32(training_data)
   48 +    }
   49 +
   50 +    fn new_coreset(&self, dimension: usize, capacity: usize) -> Self::Coreset {
   51 +        WeightedCoreset::new(dimension, capacity)
   52 +    }
   53 +
   54 +    fn append_local_coreset(
   55 +        &self,
   56 +        coreset: &mut Self::Coreset,
   57 +        data: &FixedSizeListArray,
   58 +        local_k: usize,
   59 +        max_iters: usize,
   60 +        on_progress: KMeansProgressCallback,
   61 +    ) -> Result<()> {
   62 +        append_local_coreset(
   63 +            coreset,
   64 +            data,
   65 +            self.metric_policy,
   66 +            local_k,
   67 +            max_iters,
   68 +            on_progress,
   69 +        )
   70 +    }
   71 +
   72 +    fn append_coreset(&self, coreset: &mut Self::Coreset, other: Self::Coreset) -> Result<()> {
   73 +        coreset.append(other);
   74 +        Ok(())
   75 +    }
   76 +
   77 +    fn reduce_coreset(
   78 +        &self,
   79 +        coreset: &mut Self::Coreset,
   80 +        dimension: usize,
   81 +        budget: usize,
   82 +    ) -> Result<()> {
   83 +        coreset.reduce_to_budget(dimension, budget, self.metric_policy)
   84 +    }
   85 +
   86 +    fn coreset_len(&self, coreset: &Self::Coreset) -> usize {
   87 +        coreset.len()
   88 +    }
   89 +
   90 +    fn train_centroids(
   91 +        &self,
   92 +        coreset: Self::Coreset,
   93 +        dimension: usize,
   94 +        num_partitions: usize,
   95 +        max_iters: usize,
   96 +        on_progress: KMeansProgressCallback,
   97 +    ) -> Result<FixedSizeListArray> {
   98 +        let (coreset_data, coreset_weights, coreset_losses) = coreset.into_fsl_parts(dimension)?;
   99 +        let mut centroids = {
  100 +            let params = WeightedHierarchicalKMeansParams {
  101 +                dimension,
  102 +                target_k: num_partitions,
  103 +                metric_policy: self.metric_policy,
  104 +                max_iters,
  105 +                on_progress: on_progress.clone(),
  106 +            };
  107 +            train_weighted_hierarchical_f32_kmeans(
  108 +                &coreset_data,
  109 +                &coreset_weights,
  110 +                &coreset_losses,
  111 +                &params,
  112 +            )?
  113 +        };
  114 +        let refined = refine_weighted_f32_kmeans(
  115 +            &coreset_data,
  116 +            &coreset_weights,
  117 +            &coreset_losses,
  118 +            &centroids,
  119 +            self.metric_policy,
  120 +            3,
  121 +            on_progress,
  122 +        )?;
  123 +        centroids = f32_fsl_from_values(refined.centroids, dimension)?;
  124 +        Ok(centroids)
  125 +    }

  146 +    }

  173 +    }
